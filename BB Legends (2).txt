local function Unlock()
    if busy or unlocked then return end
    busy = true

    if input.Text == REQUIRED_KEY then
        unlocked = true
        status.Text = "Access granted. Loading..."
        status.TextColor3 = Color3.fromRGB(52, 211, 153)
        unlock.Text = "ACCESS GRANTED"

        task.wait(0.25)

        pcall(function()
            local containers = {}

            local okHui, hui = pcall(function()
                return gethui and gethui()
            end)
            if okHui and hui then
                table.insert(containers, hui)
            end

            local okCore, coreGui = pcall(function()
                return game:GetService("CoreGui")
            end)
            if okCore and coreGui then
                table.insert(containers, coreGui)
            end

            local okPlayerGui, playerGui = pcall(function()
                return LocalPlayer:FindFirstChildOfClass("PlayerGui")
            end)
            if okPlayerGui and playerGui then
                table.insert(containers, playerGui)
            end

            for _, container in ipairs(containers) do
                for _, gui in ipairs(container:GetChildren()) do
                    local guiName = string.lower(tostring(gui.Name))
                    if guiName == "rayfield"
                        or guiName:find("rayfield", 1, true)
                        or guiName == "plesnevi yxurayfield"
                        or guiName == "plesnevi yxukeygate" then
                        pcall(function()
                            gui:Destroy()
                        end)
                    end
                end
            end
        end)

        keyGui:Destroy()
    else
        Deny()
        busy = false
    end
end

titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then
        local delta = input.Position - dragStart
        local newPos = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )

        mainFrame.Position = newPos

        for i, layer in ipairs(glowLayers) do
            local offset = glowSizes[i] / 2
            layer.Position = UDim2.new(
                newPos.X.Scale, newPos.X.Offset - offset,
                newPos.Y.Scale, newPos.Y.Offset - offset
            )
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

-- Original shot-meter visual adjustment
local function forcePerfectShot()
    if not enabled then return end

    local pgui = player:FindFirstChild("PlayerGui")
    if not pgui then return end

    local visual = pgui:FindFirstChild("Visual")
        or pgui:FindFirstChildWhichIsA("ScreenGui")

    if visual then
        local shooting = visual:FindFirstChild("Shooting")

        if shooting then
            local bar = shooting:FindFirstChild("Bar")
                or shooting:FindFirstChild("Meter")

            if bar and bar:IsA("Frame") then
                bar.Size = UDim2.fromScale(1, 1)

                TweenService:Create(bar, TweenInfo.new(0.005), {
                    Size = UDim2.fromScale(1, 1)
                }):Play()
            end
        end
    end

    pcall(function()
        local char = player.Character
        if char then
            for _, obj in pairs(char:GetDescendants()) do
                local n = obj.Name:lower()

                if n:find("contest") or n:find("smother")
                    or n:find("shot") or n:find("meter")
                    or n:find("release") or n:find("range")
                    or n:find("quality") or n:find("perfect")
                    or n:find("great") or n:find("late")
                    or n:find("early") then

                    if obj:IsA("NumberValue")
                        or obj:IsA("IntValue")
                        or obj:IsA("FloatValue") then
                        obj.Value = perfectValue
                    elseif obj:IsA("BindableEvent") then
                        obj:Fire(perfectValue)
                    end
                end
            end
        end
    end)
end

RunService.Heartbeat:Connect(forcePerfectShot)

UserInputService.InputBegan:Connect(function(input)

        or input.UserInputType == Enum.UserInputType.Touch
    ) then
        for i = 1, 8 do
            task.spawn(function()
                task.wait(0.002 * i)
                forcePerfectShot()
            end)
        end
    end
end)

-- Original auto-guard behavior
local guardConnection

local function startAutoGuard()
    if guardConnection then
        guardConnection:Disconnect()
    end

    guardConnection = RunService.Heartbeat:Connect(function()
        if not autoGuardEnabled then return end

        local myChar = player.Character
        if not myChar then return end

        local myRoot = myChar:FindFirstChild("HumanoidRootPart")
        if not myRoot then return end

        local closest, minDist = nil, math.huge

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                local root = plr.Character:FindFirstChild("HumanoidRootPart")

                if root then
                    local hasBall = plr.Character:FindFirstChild("Basketball")
                        or plr.Character:FindFirstChildWhichIsA("Tool")

                    if hasBall then
                        local dist = (myRoot.Position - root.Position).Magnitude

                        if dist < minDist and dist < 60 then
                            minDist = dist
                            closest = root
                        end
                    end
                end
            end
        end

        if closest then
            local direction = closest.Position - myRoot.Position

            if direction.Magnitude > 0 then
                myRoot.CFrame = CFrame.lookAt(
                    myRoot.Position,
                    myRoot.Position + direction
                )
            end
        end
    end)
end
